OVERVIEW:
This algorithm demonstrates effective organization through modular design using
FUNCTIONS. Each function has a single responsibility, making the code easy to
understand, maintain, and modify. The algorithm uses REPETITION for input
validation, SELECTION for business logic, and COMPOSITION to coordinate tasks.

MAINTAINABILITY FEATURES:
- Fee amounts (25, 0.05, 50, 40, 1000, 65) can be changed in one location
- Each function is independent and can be modified without affecting others
- Validation logic is separated from calculation logic

- Functions can be used in other programs

// =============================================================================
// FUNCTION 1: get_customer_input()
// Purpose: Collect and validate all customer booking information
// Reusability: Can be used in any booking system requiring similar data
// Maintainability: Validation rules easily modified without affecting other functions
// =============================================================================
FUNCTION get_customer_input():
    Display "ENTER CUSTOMER DETAILS"
    
    // Get customer name - no validation required (any text acceptable)
    // EFFICIENCY: No unnecessary validation reduces processing time
    Get customer_name
    
    // -------------------------------------------------------------------------
    // REPETITION + SELECTION: Validate event type
    // WHY: Ensures only valid business values are accepted
    // EFFICIENCY: Prevents invalid data from entering the system
    // MAINTAINABILITY: Easy to add new event types by modifying the IF condition
    // -------------------------------------------------------------------------
    Set valid_event = False
    
    // REPETITION: Loop continues until valid input received
    // This enhances RELIABILITY - program won't crash from bad input
    WHILE valid_event is False:
        Get event_type
        
        // Standardize input for case-insensitive comparison
        // READABILITY: User can type "concert", "CONCERT", or "Concert"
        Capitalize event_type
        
        // SELECTION: Check against allowed values
        // Using OR operator for clarity - easy to understand business rule
        IF event_type is "Concert" OR "Sports" OR "Conference":
            Set valid_event = True  // Exit loop - valid input
        ELSE:
            Display "Invalid event type. Please enter Concert, Sports, or Conference."
            // Loop continues - asks again (REPETITION enhances user experience)
        END IF
    END WHILE
    
    // -------------------------------------------------------------------------
    // REPETITION + SELECTION + ERROR HANDLING: Validate number of tickets
    // WHY: Business rule requires positive ticket count
    // EFFICIENCY: Try-catch prevents program crash from invalid input types
    // READABILITY: Clear separation of type validation and value validation
    // MAINTAINABILITY: Validation threshold (> 0) easy to modify if needed
    // -------------------------------------------------------------------------
    Set valid_tickets = False
    
    WHILE valid_tickets is False:
        TRY:
            // Attempt to convert input to integer
            // ERROR HANDLING: Catches non-numeric input
            Get num_tickets as integer
            
            // SELECTION: Validate business rule (must be positive)
            // Using > 0 (not >= 1) for clarity and readability
            IF num_tickets > 0:
                Set valid_tickets = True  // Exit loop - valid input
            ELSE:
                Display "Number of tickets must be greater than 0!"
                // Loop continues automatically (REPETITION)
            END IF
        CATCH error:
            // Handle type conversion errors gracefully
            Display "Please enter a valid whole number!"
            // EFFICIENCY: Program continues instead of crashing
        END TRY
    END WHILE
    
    // -------------------------------------------------------------------------
    // REPETITION + SELECTION: Validate ticket price
    // WHY: Prices must be positive monetary values
    // EFFICIENCY: Using float allows decimal prices (e.g., 99.99)
    // MAINTAINABILITY: Easy to add minimum/maximum price limits if needed
    // -------------------------------------------------------------------------
    Set valid_price = False
    
    WHILE valid_price is False:
        TRY:
            Get ticket_price as float  // Float allows decimal values
            
            IF ticket_price > 0:
                Set valid_price = True
            ELSE:
                Display "Price must be greater than $0!"
            END IF
        CATCH error:
            Display "Please enter a valid price (e.g., 150.00)!"
        END TRY
    END WHILE
    
    // -------------------------------------------------------------------------
    // REPETITION + SELECTION: Validate customer age
    // WHY: Age must be realistic (0-120 covers all human ages)
    // READABILITY: Using AND operator clearly shows both boundaries
    // MAINTAINABILITY: Age range (0-120) easily adjustable
    // -------------------------------------------------------------------------
    Set valid_age = False
    
    WHILE valid_age is False:
        TRY:
            Get customer_age as integer
            
            // SELECTION: Check both lower and upper bounds
            // Using AND operator enhances readability
            IF customer_age >= 0 AND customer_age <= 120:
                Set valid_age = True
            ELSE:
                Display "Please enter a valid age between 0 and 120!"
            END IF
        CATCH error:
            Display "Please enter a valid whole number for age!"
        END TRY
    END WHILE
    
    // REUSABILITY: Return all collected data for use by other functions
    // COMPOSITION: This data will be passed to calculate_fees()
    RETURN customer_name, event_type, num_tickets, ticket_price, customer_age
END FUNCTION


// =============================================================================
// FUNCTION 2: calculate_fees()
// Purpose: Calculate protection fees based on business rules
// MAINTAINABILITY: Fee amounts stored as constants - easy to change in one place
// EFFICIENCY: Separate calculation function allows reuse without re-validation
// READABILITY: Step-by-step calculation is easy to follow and verify
// =============================================================================
FUNCTION calculate_fees(num_tickets, ticket_price, customer_age):
    
    // STEP 1: Calculate total ticket value
    // EFFICIENCY: Single multiplication operation
    // READABILITY: Clear variable name explains what value represents
    Set total_ticket_value = num_tickets × ticket_price
    
    // STEP 2: Apply base service fee
    // MAINTAINABILITY: Constant value (25) can be changed in one location
    // This flat fee applies to all bookings
    Set base_fee = 25.00
    
    // STEP 3: Calculate protection fee (5% of total)
    // MAINTAINABILITY: Percentage (0.05) can be modified easily
    // READABILITY: Using 0.05 is clearer than 5/100
    Set protection_fee = total_ticket_value × 0.05
    
    // -------------------------------------------------------------------------
    // STEP 4: SELECTION - Determine high value surcharge
    // WHY: Business rule applies $50 surcharge when total exceeds $1000
    // EFFICIENCY: Single comparison determines surcharge application
    // MAINTAINABILITY: Threshold ($1000) and surcharge ($50) easily modified
    // READABILITY: Using > (not >=) matches business language "exceeds"
    // -------------------------------------------------------------------------
    
    // SELECTION: Check if surcharge applies
    // Using > operator (not >=) because rule says "exceeds" not "equals or exceeds"
    IF total_ticket_value > 1000:
        Set high_value_surcharge = 50.00  // Apply surcharge
    ELSE:
        Set high_value_surcharge = 0.00   // No surcharge
    END IF
    
    // -------------------------------------------------------------------------
    // STEP 5: SELECTION - Determine age surcharge
    // WHY: Business rule applies $40 surcharge for customers over 65
    // MAINTAINABILITY: Age threshold (65) and surcharge (40) easily changed
    // EFFICIENCY: Simple comparison - no complex calculations needed
    // -------------------------------------------------------------------------
    
    IF customer_age > 65:
        Set age_surcharge = 40.00   // Apply senior surcharge
    ELSE:
        Set age_surcharge = 0.00    // No age surcharge
    END IF
    
    // STEP 6: Calculate total protection fee
    // COMPOSITION: Combining all fee components
    // READABILITY: Addition clearly shows all fee components
    // MAINTAINABILITY: Easy to add new fee types by extending this calculation
    Set total_fee = base_fee + protection_fee + high_value_surcharge + age_surcharge
    
    // REUSABILITY: Return all calculated values for display or further processing
    RETURN total_ticket_value, base_fee, protection_fee, 
           high_value_surcharge, age_surcharge, total_fee
END FUNCTION


// =============================================================================
// FUNCTION 3: display_results()
// Purpose: Display fee breakdown in clear, formatted output
// READABILITY: Well-organized output enhances user experience
// MAINTAINABILITY: Output format easily modified without affecting calculations
// EFFICIENCY: Conditional display prevents showing $0.00 surcharges
// =============================================================================
FUNCTION display_results(customer_name, event_type, num_tickets, ticket_price,
                        total_ticket_value, base_fee, protection_fee,
                        high_value_surcharge, age_surcharge, total_fee):
    
    // Display formatted header
    Display "=================================="
    Display "FEE BREAKDOWN"
    Display "=================================="
    
    // Show customer information for confirmation
    Display "Customer: " + customer_name
    Display "Event: " + event_type
    Display "Tickets: " + num_tickets + " x $" + ticket_price
    Display ""
    Display "Total Ticket Value: $" + total_ticket_value
    Display "----------------------------------"
    
    // Display base charges (always applicable)
    Display "1. Base Service Fee:     $" + base_fee
    Display "2. Protection Fee (5%):  $" + protection_fee
    
    // -------------------------------------------------------------------------
    // SELECTION: Conditionally display surcharges
    // WHY: Only show surcharges that apply (enhances readability)
    // EFFICIENCY: Prevents cluttering output with $0.00 values
    // READABILITY: User only sees relevant charges
    // -------------------------------------------------------------------------
    
    // SELECTION: Show high value surcharge only if applied
    IF high_value_surcharge > 0:
        Display "3. High Value Surcharge: $" + high_value_surcharge
        Display "   (Applied: Total value exceeds $1000)"
    END IF
    
    // SELECTION: Show age surcharge only if applied
    IF age_surcharge > 0:
        Display "4. Age Surcharge:        $" + age_surcharge
        Display "   (Applied: Customer age over 65)"
    END IF
    
    // Display total with emphasis
    Display "----------------------------------"
    Display "TOTAL PROTECTION FEE:    $" + total_fee
    Display "=================================="
END FUNCTION


// =============================================================================
// FUNCTION 4: ask_continue()
// Purpose: Validate yes/no input and control program flow
// REUSABILITY: This pattern can be used for any yes/no confirmation
// EFFICIENCY: Accepts multiple valid inputs (yes/y/no/n) for user convenience
// MAINTAINABILITY: Easy to modify accepted values
// =============================================================================
FUNCTION ask_continue():
    
    // -------------------------------------------------------------------------
    // REPETITION + SELECTION: Validate yes/no input
    // WHY: Ensures program only exits or continues with valid user choice
    // READABILITY: Clear validation logic
    // EFFICIENCY: Case-insensitive comparison improves user experience
    // -------------------------------------------------------------------------
    
    Set valid_choice = False
    
    WHILE valid_choice is False:
        Get choice from user
        
        // Convert to lowercase for case-insensitive comparison
        // READABILITY: User can type "Yes", "YES", "yes", etc.
        Convert choice to lowercase
        
        // SELECTION: Check for valid affirmative responses
        // Using OR operator for multiple acceptable inputs
        IF choice is "yes" OR choice is "y":
            Set valid_choice = True
            RETURN True  // Continue processing customers
            
        // SELECTION: Check for valid negative responses
        ELSE IF choice is "no" OR choice is "n":
            Set valid_choice = True
            RETURN False  // Exit program
            
        ELSE:
            // Invalid input - show helpful message
            // READABILITY: Clear feedback guides user
            Display "Invalid input. Please enter 'yes' or 'no'"
            // Loop continues - asks again (REPETITION)
        END IF
    END WHILE
END FUNCTION


// =============================================================================
// FUNCTION 5: main()
// Purpose: Main program coordinator - demonstrates COMPOSITION
// COMPOSITION: Orchestrates all other functions to accomplish the task
// MAINTAINABILITY: Changes to individual functions don't affect coordination
// READABILITY: Clear program flow - easy to understand execution sequence
// =============================================================================
FUNCTION main():
    
    // Display welcome message for professional user experience
    Display "=================================="
    Display "EVENTSURE NZ - TICKET PROTECTION CALCULATOR"
    Display "=================================="
    
    // Initialize program control variable
    Set continue_program = True
    
    // -------------------------------------------------------------------------
    // REPETITION: Main program loop
    // WHY: Allows processing multiple customers in one session
    // EFFICIENCY: User doesn't need to restart program for each customer
    // COMPOSITION: Each iteration uses all functions in coordinated sequence
    // -------------------------------------------------------------------------
    
    WHILE continue_program is True:
        
        // STEP 1: COMPOSITION - Get customer data
        // Calls get_customer_input() which handles all validation
        // MAINTAINABILITY: Input logic separated from main flow
        Set customer_data = get_customer_input()
        
        // STEP 2: COMPOSITION - Calculate fees
        // Calls calculate_fees() with validated data
        // REUSABILITY: Calculation function independent of input method
        Set fee_data = calculate_fees(customer_data)
        
        // STEP 3: COMPOSITION - Display results
        // Calls display_results() with both input and calculated data
        // MAINTAINABILITY: Output format independent of calculation
        Call display_results(customer_data, fee_data)
        
        // STEP 4: COMPOSITION - Ask to continue
        // Calls ask_continue() to control loop execution
        // READABILITY: Clear loop control mechanism
        Set continue_program = ask_continue()
        
    END WHILE
    
    // Display goodbye message when user exits
    Display ""
    Display "=================================="
    Display "Thank you for using EventSure NZ!"
    Display "Have a great day!"
    Display "=================================="
    
END FUNCTION


// =============================================================================
// MAIN PROGRAM ENTRY POINT
// =============================================================================
MAIN PROGRAM:
    Call main()  // Start the application
END PROGRAM
